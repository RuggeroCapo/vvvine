async function changeCheckoutAddress() {
    const addressSelects = document.querySelectorAll('select[name="line-item-address"]');

    if (!addressSelects.length) {
        console.log("Nessun selettore indirizzo trovato.");
        return false;
    }

    let changed = false;

    for (const select of addressSelects) {
        if (!select || select.offsetParent === null) {
            continue;
        }

        const options = Array.from(select.options).filter(option => option.value);

        if (options.length === 0) {
            continue;
        }

        const currentValue = select.value;

        attemptedCheckoutAddresses.add(currentValue);

        let nextOption = null;

        const scaleaOption = options.find(option => {
            const text = option.textContent.trim().toLowerCase();
            return text.includes("scalea") &&
                !attemptedCheckoutAddresses.has(option.value);
        });

        if (scaleaOption) {
            nextOption = scaleaOption;
            console.log("Seleziono prioritariamente l'indirizzo Scalea.");
        } else {
            nextOption = options.find(option =>
                                      !attemptedCheckoutAddresses.has(option.value)
                                     );
        }

        if (!nextOption) {
            console.log("Tutti gli indirizzi disponibili sono già stati tentati.");
            continue;
        }

        console.log(
            "Indirizzo selezionato:",
            nextOption.textContent.trim()
        );

        select.value = nextOption.value;

        attemptedCheckoutAddresses.add(nextOption.value);

        select.dispatchEvent(
            new Event('input', {
                bubbles: true
            })
        );

        select.dispatchEvent(
            new Event('change', {
                bubbles: true
            })
        );

        changed = true;
    }

    if (!changed) {
        return false;
    }

    await new Promise(resolve => setTimeout(resolve, 500));

    const continueButton =
          document.querySelector(
              '#checkout-secondary-continue-button-id input[data-testid="secondary-continue-button"]'
          ) ||
          document.querySelector(
              '#checkout-secondary-continue-button-id input[type="submit"]'
          );

    if (!continueButton) {
        console.log("Pulsante Continua non trovato.");
        return false;
    }

    console.log("Clicco Continua.");

    continueButton.click();

    await new Promise(resolve => setTimeout(resolve, 1500));

    console.log("Indirizzo confermato.");

    return true;
}

    try{
        const snapshot = await ref.get();

        if(orderStartTime!==0){

            orderStartTime=0;
        }
    }
});

function createAddressDropdown(){
    const addressElements=document.querySelectorAll('.vvp-address-option');

    if(addressElements.length===0)return;

    const dropdown=document.createElement('select');
    dropdown.style.marginRight='10px';
    dropdown.id='address-selector';

    addressElements.forEach(element=>{
        const addressId=element.getAttribute('data-address-id');
        const legacyAddressId=element.getAttribute('data-legacy-address-id');
        const label=element.querySelector('.a-radio-label')?.textContent.trim().split('\n')[0];

        if(label){
            const option=document.createElement('option');
            option.value=addressId;
            option.textContent=label;
            option.dataset.legacyAddressId=legacyAddressId;
            dropdown.appendChild(option);
        }
    });

    if(dropdown.options.length>0){
        if(selectedAddressId){
            dropdown.value=selectedAddressId;
        }else{
            selectedAddressId=dropdown.options[0].value;
            selectedLegacyAddressId=dropdown.options[0].dataset.legacyAddressId;
        }
    }

    dropdown.onchange=function(){
        selectedAddressId=this.value;
        selectedLegacyAddressId=this.options[this.selectedIndex].dataset.legacyAddressId;
        localStorage.setItem('selectedAddressId',selectedAddressId);
        localStorage.setItem('selectedLegacyAddressId',selectedLegacyAddressId);
    };

    const container=document.querySelector('.a-section.vvp-container-right-align');

    if(container)container.prepend(dropdown);
}

async function fetchPromoId(recId,asin){
    const url=`https://${amazonDomain}/vine/api/recommendations/${encodeURIComponent(recId)}/item/${asin}?imageSize=500`;
    const response=await fetch(url);
    const data=await response.json();
    const result=data.result||data;
    return result.promotionId||result.receiptData?.promotionId||result.item?.promotionId;
}

async function fetchOfferId(recId,asin){
    const payload=JSON.stringify({
        recommendationId:recId,
        recommendationType:"VINE_FOR_ALL",
        itemAsin:asin,
        addressId:selectedAddressId,
        legacyAddressId:selectedLegacyAddressId
    });

    try{
        const response=await fetch(
            `https://${amazonDomain}/vine/api/voiceOrders`,
            {
                method:'POST',
                body:payload,
                headers:{
                    'anti-csrftoken-a2z':CRSRF_TOKEN,
                    'content-type':'application/json'
                }
            }
        );

        const text=await response.text();

        let offerId=(text.match(/"offer(?:Listing)?Id"\s*:\s*"(.*?)"/i)||[])[1];

        if(offerId){
            return{
                success:true,
                id:offerId
            };
        }

        const j=JSON.parse(text);
        const errorMsg=j.error||(j.result&&j.result.error);

        if(errorMsg){
            return{
                success:false,
                error:errorMsg
            };
        }
    }catch(e){
        console.error("API Parse Error",e);
    }

    return{
        success:false,
        error:"No Offer ID found"
    };
}

async function performOrder(recId,asin,isParent){
    if(!selectedAddressId||!CRSRF_TOKEN)return;

    attemptedCheckoutAddresses.clear();
    checkoutErrorReported = false;

    let finalAsin=asin;
    let realRecId=recId;

    try{
        if(isParent){
            const url=`https://${amazonDomain}/vine/api/recommendations/${encodeURIComponent(recId)}`;
            const response=await fetch(url);
            const data=await response.json();
            const result=data.result||data;

            if(result.variations&&result.variations.length>0){
                finalAsin=result.variations[0].asin;
                realRecId=result.recommendationId||recId;
            }
        }

        const [promoId,offerResult]=await Promise.all([
            fetchPromoId(realRecId,finalAsin),
            fetchOfferId(realRecId,finalAsin)
        ]);

        if(offerResult.success&&offerResult.id&&promoId){
            const form=document.createElement('form');
            form.method='POST';
            form.action='/checkout/entry/buynow?pipelineType=Chewbacca';
            form.style.display='none';
            form.target='vine_checkout_frame';

            const inputs={
                skipCart:'1',
                quantity:'1',
                asin:finalAsin,
                offerListingID:offerResult.id,
                vinePromotionType:'VINE',
                vinePromotionId:promoId
            };

            for(const key in inputs){
                const inp=document.createElement('input');
                inp.type='hidden';
                inp.name=key;
                inp.value=inputs[key];
                form.appendChild(inp);
            }

            document.body.appendChild(form);
            updateOverlayMessage("Navigating...");
            form.submit();
        }else{
            let msg="Error: Missing Data";

            if(!offerResult.success&&offerResult.error){
                msg="Error: "+offerResult.error.replace(/_/g,' ');
            }

            updateOverlayMessage(msg);

            setTimeout(()=>{
                const overlay=document.getElementById('vine-overlay');
                if(overlay)overlay.style.display='flex';
            },3000);
        }
    }catch(e){
        console.error("Order failed",e);
        updateOverlayMessage("Error!");

        setTimeout(()=>{
            const overlay=document.getElementById('vine-overlay');
            if(overlay)overlay.style.display='flex';
        },3000);
    }
}
